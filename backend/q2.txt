-- Simple query: report student results with exam title
SELECT
    r.id,
    u.username,
    e.title AS exam_title,
    r.score,
    r.total_questions,
    ROUND((r.score::numeric / NULLIF(r.total_questions, 0)) * 100, 2) AS percentage,
    r.date_taken
FROM results04 r
JOIN users04 u ON r.user_id = u.id
JOIN exams04 e ON r.exam_id = e.id
ORDER BY r.date_taken DESC;
